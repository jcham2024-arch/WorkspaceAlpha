// Good: Immutable data
data class User(
    val id: String,
    val name: String,
    val email: String,
)

// Good: Transform with copy()
fun updateEmail(user: User, newEmail: String): User =
    user.copy(email = newEmail)

val users: List<User> = listOf(user1, user2)
val filtered = users.filter { it.email.isNotBlank() }
